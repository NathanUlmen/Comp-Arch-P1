0x84 -> 0xFFF       // End of List
0x85 -> 0x080       // Beginning entry location

//--------- Beginning of File ---------//

// PROGRAM SET UP
LDR 0x00 R6         // Load constant 1 into R6
LDR 0x01 R7         // Load constant 2 into R7
LDR 0x02 R8         // Load constant 3 into R8

LDR 0X03 R14        // Load constant 12 into R14
ADD R14 PC R14      // Load location of HALT to R14

LDR 0x04 R15        // Load end of file token into R15

LDR 0x85 R10        // Load address of first entry into R9

LDI 0x85 R0         // Load the first value into R0

ADD R5 PC R6        // Load location of next instruction into R5

// LOOP
ADD R10 R10 R6      // Get address of next entry and store in R10
STR 0x86 R10        // Store address of next entry at location 0x86
LDI 0x86 R1         // Load the next value into R1

BEQ R14 R1 R15      // Check if all the comparisons have been made

BLT R5 R0 R1        // Go back to the beginning of the loop if R0 is less than R1

// R1 < R0
MOV R0 R1
